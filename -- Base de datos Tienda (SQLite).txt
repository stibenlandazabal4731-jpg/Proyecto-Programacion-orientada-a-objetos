-- Base de datos: Tienda (SQLite)
-- Ejecutar en DB Browser for SQLite, DBeaver o con: sqlite3 tienda.db < tienda_sqlite.sql

PRAGMA foreign_keys = ON;

DROP TABLE IF EXISTS inventario;
DROP TABLE IF EXISTS ventas;
DROP TABLE IF EXISTS productos;
DROP TABLE IF EXISTS usuarios;

CREATE TABLE usuarios (
    id_usuario INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre     TEXT NOT NULL,
    correo     TEXT NOT NULL UNIQUE,
    rol        TEXT NOT NULL
);

CREATE TABLE productos (
    id_producto   INTEGER PRIMARY KEY AUTOINCREMENT,
    codigo        TEXT NOT NULL UNIQUE,
    nombre        TEXT NOT NULL,
    unidad_medida TEXT NOT NULL,
    precio        NUMERIC NOT NULL CHECK (precio >= 0),
    stock_actual  INTEGER NOT NULL CHECK (stock_actual >= 0),
    stock_minimo  INTEGER NOT NULL DEFAULT 5 CHECK (stock_minimo >= 0)
);

CREATE TABLE ventas (
    id_venta    INTEGER PRIMARY KEY AUTOINCREMENT,
    id_usuario  INTEGER REFERENCES usuarios(id_usuario),
    fecha_venta TEXT DEFAULT (datetime('now', 'localtime')),
    total_venta NUMERIC NOT NULL CHECK (total_venta >= 0)
);

CREATE TABLE inventario (
    id_inventario    INTEGER PRIMARY KEY AUTOINCREMENT,
    id_venta         INTEGER REFERENCES ventas(id_venta) ON DELETE CASCADE,
    id_producto      INTEGER REFERENCES productos(id_producto),
    cantidad         INTEGER NOT NULL CHECK (cantidad > 0),
    subtotal         NUMERIC NOT NULL CHECK (subtotal >= 0),
    fecha_movimiento TEXT DEFAULT (datetime('now', 'localtime'))
);

-- Consultas de prueba
SELECT * FROM productos;

SELECT * FROM usuarios;

-- Productos con stock bajo
SELECT codigo, nombre, stock_actual, stock_minimo
FROM productos
WHERE stock_actual <= stock_minimo;

-- Detalle de ventas
SELECT
    i.id_inventario,
    v.id_venta,
    u.nombre AS cajero,
    p.nombre AS producto,
    p.unidad_medida,
    i.cantidad,
    i.subtotal,
    v.total_venta,
    i.fecha_movimiento
FROM inventario i
JOIN ventas v    ON i.id_venta = v.id_venta
JOIN productos p ON i.id_producto = p.id_producto
JOIN usuarios u  ON v.id_usuario = u.id_usuario;
